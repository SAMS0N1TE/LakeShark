# BLE EXPLORER flows. The app has these built in (bx_flows_builtin.c holds
# the same text); PACKS can put a newer copy at
# /sdcard/lakeshark/packs/ble-flows.bin, and your own files go in
# /sdcard/lakeshark/ble-flows/*.flow. The format is in the app's README.
#
# Every flow is a short, fixed sequence you start yourself. A flow that
# writes asks first.

# ---- Bluetooth SIG services, written from the public SIG specifications ----

flow Device information
about Reads maker, model, serial and firmware (Device Information)
match service 180a
# Bluetooth SIG Device Information Service 1.1
? read 2a29
? read 2a24
? read 2a25
? read 2a27
? read 2a26
? read 2a28
? read 2a23
? read 2a50
? read 2a2a

flow Device name and look
about Reads the name it gives itself and what kind of device it says it is
match service 1800
# Bluetooth SIG Generic Access (Core Specification, Vol 3 Part C)
? read 2a00
? read 2a01
? read 2a04

flow Battery level
about Reads the battery level and listens 5 s for a change
match service 180f
# Bluetooth SIG Battery Service 1.0
read 2a19
? notify 2a19 on
? wait notify 2a19 5000
? notify 2a19 off

flow Find me: beep
about Asks a tag or tracker to beep (Immediate Alert, high) for 5 s
confirm Makes the device beep for 5 seconds. Run it?
match service 1802
# Bluetooth SIG Immediate Alert Service / Find Me Profile.
# Alert Level only takes writes without response; see the README.
write 1802/2a06 02
wait 5000
write 1802/2a06 00

flow Find me: quiet
about Stops a beep started with Find me (Immediate Alert, none)
confirm Sends "no alert" to the device. Run it?
match service 1802
# Bluetooth SIG Immediate Alert Service
write 1802/2a06 00

flow Current time: read
about Reads the clock of a watch, band or sensor
match service 1805
# Bluetooth SIG Current Time Service 1.1
read 2a2b
? read 2a0f

flow Current time: set
about Sets the device clock to this board's time
confirm Sets the device clock to the board's time. Run it?
match service 1805
# Bluetooth SIG Current Time Service 1.1: date, time, weekday, 1/256 s,
# reason 01 (set by hand). Many devices make this read only.
write 2a2b $time
read 2a2b

flow Heart rate: listen
about Turns on heart rate for about 10 s, then off again
match service 180d
# Bluetooth SIG Heart Rate Service 1.0
? read 2a38
notify 2a37 on
? wait notify 2a37 5000
? wait notify 2a37 3000
? wait notify 2a37 3000
notify 2a37 off

flow Fitness machine: listen
about Reads what the machine supports, then its live data for 6 s

# ---- serial bridges ----

flow Nordic UART: ping
about Sends a newline over Nordic UART and listens 3 s for a reply
match service 6e400001-b5a3-f393-e0a9-e50e24dcca9e
# Written from Nordic Semiconductor's public Nordic UART Service docs:
# 6e400002 RX (write), 6e400003 TX (notify).
notify 6e400003-b5a3-f393-e0a9-e50e24dcca9e on
write 6e400002-b5a3-f393-e0a9-e50e24dcca9e "\r\n"
? wait notify 6e400003-b5a3-f393-e0a9-e50e24dcca9e 3000
notify 6e400003-b5a3-f393-e0a9-e50e24dcca9e off

flow HM-10 serial: ping
about Sends AT over an HM-10, HC-08 or JDY style module, listens 3 s
match service ffe0
match name HMSoft*
match name HC-08*
match name JDY-*
# Written from the public HM-10 data sheet: service ffe0, characteristic
# ffe1 carries the serial data both ways. Connected, the module passes the
# bytes to its host; many hosts answer AT with OK.
notify ffe0/ffe1 on
write ffe0/ffe1 "AT\r\n"
? wait notify ffe0/ffe1 3000
notify ffe0/ffe1 off

flow Microchip UART: ping
about Sends a newline over Microchip Transparent UART, listens 3 s
match service 49535343-fe7d-4ae5-8fa9-9fafd205e455
# Written from Microchip's public BM70/RN4870 Transparent UART docs:
# 1e4d TX (notify), 8841 RX (write).
notify 49535343-1e4d-4bd9-ba61-23c647249616 on
write 49535343-8841-43f4-a8d4-ecbe34729bb3 "\r\n"
? wait notify 49535343-1e4d-4bd9-ba61-23c647249616 3000
notify 49535343-1e4d-4bd9-ba61-23c647249616 off

# ---- printers ----

flow Thermal printer: status
about Asks an ESC/POS receipt printer for its status byte
match service 18f0
match name *Printer*
match name MPT-*
match name PT-2*
# Written from the public Epson ESC/POS reference: DLE EOT 1, printer
# status. Common Bluetooth receipt printers take ESC/POS on 18f0/2af1 and
# answer on 2af0.
? notify 2af0 on
write 2af1 10 04 01
? wait notify 2af0 2000
? notify 2af0 off

flow Thermal printer: test print
about Prints two short lines on an ESC/POS receipt printer
confirm Prints two lines of text on the printer. Run it?
match service 18f0
match name *Printer*
match name MPT-*
match name PT-2*
# Written from the public Epson ESC/POS reference: ESC @ resets, ESC d n
# feeds n lines.
write 2af1 1b 40
write 2af1 "LakeShark\n"
write 2af1 "BLE EXPLORER test\n"
write 2af1 1b 64 03

flow Cat printer: status
about Asks a GB01, GB02, MX05 style cat printer for its state
match service ae30
match name GB0*
match name MX0*
match name MX1*
match name YT01
# Ported from rbaron/catprinter (MIT): 51 78 <cmd> 00 <len> 00 <data>
# <crc8> ff on ae01, replies on ae02. a3 is get device state.
notify ae02 on
write ae01 51 78 a3 00 01 00 00 00 ff
? wait notify ae02 3000
notify ae02 off

flow Cat printer: feed paper
about Feeds a little paper out of a cat printer
confirm Feeds the printer's paper. Run it?
match service ae30
match name GB0*
match name MX0*
match name MX1*
match name YT01
# Ported from rbaron/catprinter (MIT): bd feeds n dot lines.
write ae01 51 78 bd 00 01 00 40 c7 ff

flow Cat printer: test print
about Prints four black rows and a dashed line, then feeds the paper
confirm Prints a small test pattern on the cat printer. Run it?
match service ae30
match name GB0*
match name MX0*
match name MX1*
match name YT01
# Ported from rbaron/catprinter (MIT), cmds_print_img: state, 200 dpi,
# energy, apply, lattice start, run-length rows (bf), feed, paper, lattice
# end. Rows are 384 dots: ff ff ff 83 is 384 black; a0 20 is 32 black,
# 32 white.
notify ae02 on
write ae01 51 78 a3 00 01 00 00 00 ff
write ae01 51 78 a4 00 01 00 32 9e ff
write ae01 51 78 af 00 02 00 2e e0 d6 ff
write ae01 51 78 be 00 01 00 01 07 ff
write ae01 51 78 a6 00 0b 00 aa 55 17 38 44 5f 5f 5f 44 38 2c a1 ff
write ae01 51 78 bf 00 04 00 ff ff ff 83 ad ff
write ae01 51 78 bf 00 04 00 ff ff ff 83 ad ff
write ae01 51 78 bf 00 04 00 ff ff ff 83 ad ff
write ae01 51 78 bf 00 04 00 ff ff ff 83 ad ff
write ae01 51 78 bf 00 0c 00 a0 20 a0 20 a0 20 a0 20 a0 20 a0 20 17 ff
write ae01 51 78 bf 00 0c 00 a0 20 a0 20 a0 20 a0 20 a0 20 a0 20 17 ff
write ae01 51 78 bf 00 0c 00 a0 20 a0 20 a0 20 a0 20 a0 20 a0 20 17 ff
write ae01 51 78 bf 00 0c 00 a0 20 a0 20 a0 20 a0 20 a0 20 a0 20 17 ff
write ae01 51 78 bd 00 01 00 19 4f ff
write ae01 51 78 a1 00 02 00 30 00 f9 ff
write ae01 51 78 a6 00 0b 00 aa 55 17 00 00 00 00 00 00 00 17 11 ff
write ae01 51 78 a3 00 01 00 00 00 ff
? wait notify ae02 2000
notify ae02 off

# ---- other devices ----

flow WeChat AirSync: probe
about Reads the MAC and listens for the auth request. Writes nothing
match service fee7
# Written from the public WeChat hardware (AirSync) protocol: fec7 write,
# fec8 indicate, fec9 read MAC. Turning on fec8 makes the device send its
# auth request (fe 01 <len> 27 11 <seq>); the log decodes it.
? read fec9
notify fec8 on
? wait notify fec8 5000
? wait notify fec8 2000
notify fec8 off

flow Xiaomi thermometer: listen
about Waits up to 10 s for temperature, humidity and battery mV
match service ebe0ccb0-7a0a-4b0c-8a1a-6ff2997da3a6
match name LYWSD03MMC
# Written from public notes on the LYWSD03MMC: ebe0ccc1 notifies about
# every 6 s: int16 temperature x0.01 C, humidity %, uint16 battery mV.
notify ebe0ccc1-7a0a-4b0c-8a1a-6ff2997da3a6 on
? wait notify ebe0ccc1-7a0a-4b0c-8a1a-6ff2997da3a6 10000
notify ebe0ccc1-7a0a-4b0c-8a1a-6ff2997da3a6 off

flow Mi Flora: sensors
about Reads battery and firmware, then temperature, light and moisture
confirm Writes a0 1f to switch the plant sensor to live readings. Run it?
match service 1204
match name Flower care
match name Flower mate
# Ported from open-homeautomation/miflora (MIT): handles 0x38, 0x33 and
# 0x35 are 1a02 (battery, firmware), 1a00 (mode) and 1a01 (sensor data).
read 1a02
write 1a00 a0 1f
read 1a01

flow Govee light: on
about Turns a Govee H6xxx light on
confirm Turns the light on. Run it?
match service 00010203-0405-0607-0809-0a0b0c0d1910
match name ihoment_H*
match name Govee_H*
# Ported from Beshelmek/govee_ble_lights (MIT): 33 <cmd> <data>, padded to
# 19 bytes, then the XOR of all of them. Command 01 is power.
write 00010203-0405-0607-0809-0a0b0c0d2b11 33 01 01 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 00 33

flow Triones light: on
about Turns a Triones or HappyLighting LED strip on
confirm Turns the LED strip on. Run it?
match service ffd5
match name Triones*
# Written from the publicly documented Triones protocol (no licensed code
# used): service ffd5, write ffd9, cc 23 33 on, cc 24 33 off.
write ffd5/ffd9 cc 23 33

flow Triones light: off
about Turns a Triones or HappyLighting LED strip off
confirm Turns the LED strip off. Run it?
match service ffd5
match name Triones*
# Written from the publicly documented Triones protocol.
write ffd5/ffd9 cc 24 33
