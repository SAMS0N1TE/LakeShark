#include <stdio.h>
#include <string.h>
#include "mt_aes.h"
#include "mt_proto.h"

static int fails;
static void check(int ok, const char *what) { printf("%s  %s\n", ok ? "ok  " : "FAIL", what); fails += !ok; }
static void hex(const char *s, uint8_t *out) { for (int i = 0; s[2 * i]; i++) sscanf(s + 2 * i, "%2hhx", &out[i]); }

int main(void)
{
    uint8_t key[32], pt[16], ct[16], out[16];
    mt_aes_t a;
    /* FIPS-197 C.1 AES-128 */
    hex("000102030405060708090a0b0c0d0e0f", key); hex("00112233445566778899aabbccddeeff", pt);
    hex("69c4e0d86a7b0430d8cdb78070b4c55a", ct);
    mt_aes_setkey(&a, key, 16); mt_aes_block(&a, pt, out);
    check(!memcmp(out, ct, 16), "AES-128 FIPS-197 C.1");
    /* FIPS-197 C.3 AES-256 */
    hex("000102030405060708090a0b0c0d0e0f101112131415161718191a1b1c1d1e1f", key);
    hex("8ea2b7ca516745bfeafc49904b496089", ct);
    mt_aes_setkey(&a, key, 32); mt_aes_block(&a, pt, out);
    check(!memcmp(out, ct, 16), "AES-256 FIPS-197 C.3");
    /* SP 800-38A F.5.1 CTR-AES128 first two blocks */
    uint8_t iv[16], buf[32], exp[32];
    hex("2b7e151628aed2a6abf7158809cf4f3c", key); hex("f0f1f2f3f4f5f6f7f8f9fafbfcfdfeff", iv);
    hex("6bc1bee22e409f96e93d7e117393172aae2d8a571e03ac9c9eb76fac45af8e51", buf);
    hex("874d6191b620e3261bef6864990db6ce9806f66b7970fdff8617187bb9fffdff", exp);
    mt_aes_setkey(&a, key, 16); mt_aes_ctr(&a, iv, buf, 32);
    check(!memcmp(buf, exp, 32), "AES-CTR SP 800-38A F.5.1");

    int slot, slots;
    uint64_t f = mt_frequency_hz(0, 0, "", &slot, &slots);
    printf("     US LongFast %llu Hz slot %d/%d\n", (unsigned long long)f, slot, slots);
    check(f == 906875000ull && slot == 20, "US LongFast = 906.875 MHz slot 20 (published)");
    f = mt_frequency_hz(2, 0, "", &slot, &slots);
    check(f == 869525000ull, "EU_868 LongFast = 869.525 MHz (published)");
    f = mt_frequency_hz(1, 1, "", &slot, &slots);
    printf("     ANZ MediumFast %llu Hz slot %d/%d\n", (unsigned long long)f, slot, slots);

    uint8_t k[32]; const uint8_t one = 1;
    int kl = mt_channel_key(&one, 1, k);
    printf("     default key len %d, LongFast hash 0x%02x, MediumFast hash 0x%02x\n", kl,
           mt_channel_hash("LongFast", k, kl), mt_channel_hash("MediumFast", k, kl));
    check(mt_channel_hash("LongFast", k, kl) == 0x08, "LongFast + default key channel hash = 0x08 (published as 8)");

    /* round trip a text packet */
    uint8_t pl[64]; const char *msg = "hola desde LakeShark";
    mt_data_t d = { MT_PORT_TEXT, (const uint8_t *)msg, strlen(msg), false };
    int n = mt_data_encode(&d, pl, sizeof(pl));
    mt_crypt(k, kl, 0x11223344, 0xdeadbeef, pl, (size_t)n);
    mt_crypt(k, kl, 0x11223344, 0xdeadbeef, pl, (size_t)n);
    mt_data_t back;
    check(mt_data_decode(pl, (size_t)n, &back) && back.portnum == 1 && back.payload_len == strlen(msg) &&
          !memcmp(back.payload, msg, back.payload_len), "Data encode/encrypt/decrypt/decode round trip");
    mt_header_t h = { MT_BROADCAST, 0x11223344, 0xdeadbeef, 3, 3, 0, 0, 0x08, 0, 0x44 }, h2;
    uint8_t hb[16]; mt_header_pack(&h, hb); mt_header_unpack(hb, 16, &h2);
    check(!memcmp(&h, &h2, sizeof(h)) && hb[12] == 0x63, "header pack/unpack (flags 0x63 for hop 3/3)");
    mt_user_t u = { "!11223344", "LakeShark LR", "LSLR", 255, 1 }, u2;
    uint8_t mac[6] = {1, 2, 3, 4, 5, 6};
    n = mt_user_encode(&u, mac, pl, sizeof(pl));
    check(n > 0 && mt_user_decode(pl, (size_t)n, &u2) && !strcmp(u2.long_name, "LakeShark LR") && u2.hw_model == 255,
          "User encode/decode");
    printf("%s\n", fails ? "SOME FAILED" : "ALL PASS");
    return fails;
}
